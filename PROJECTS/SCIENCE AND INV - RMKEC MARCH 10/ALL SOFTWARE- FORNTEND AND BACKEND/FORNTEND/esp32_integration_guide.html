ESP32 Firmware — Send JSON over Wi-Fi
ADD THIS TO YOUR EXISTING ESP32 CODE
Arduino / ESP32WiFi.hHTTPClient.h
How it works: Your ESP32 connects to Wi-Fi and after every sensor reading, it sends a JSON POST request to your laptop's Flask server. Your laptop and ESP32 must be on the same Wi-Fi network.
C++ / Arduino// ─── ADD THESE INCLUDES AT THE TOP OF YOUR FIRMWARE ───────────
#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>   // Install from Library Manager

// ─── YOUR WI-FI CREDENTIALS ────────────────────────────────────
const char* ssid     = "YOUR_WIFI_NAME";
const char* password = "YOUR_WIFI_PASSWORD";

// ─── YOUR LAPTOP'S IP ON THE SAME WI-FI ────────────────────────
// Find it: Windows → ipconfig | Mac/Linux → ifconfig
const char* serverUrl = "http://192.168.1.5:5001/data";
//                              ↑ CHANGE THIS TO YOUR LAPTOP IP

// ─── CONNECT TO WI-FI IN setup() ───────────────────────────────
void setup() {
  Serial.begin(115200);
  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nConnected! IP: " + WiFi.localIP().toString());
  // ... rest of your setup code
}

// ─── SEND SENSOR DATA FUNCTION ─────────────────────────────────
void sendDataToServer() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  http.begin(serverUrl);
  http.addHeader("Content-Type", "application/json");

  // Build JSON with your sensor variables
  StaticJsonDocument<512> doc;
  doc["aqi"]        = aqiValue;        // your existing variable
  doc["co2"]        = co2Value;
  doc["temp"]       = temperature;
  doc["humidity"]   = humidity;
  doc["nh3"]        = nh3Value;
  doc["benzene"]    = benzeneValue;
  doc["co"]         = coValue;
  doc["voltage"]    = busVoltage;
  doc["current"]    = currentMA;
  doc["power"]      = powerW;
  doc["energy"]     = energyKWh;
  doc["cost"]       = costValue;
  doc["occupancy"]  = occupancyState;  // 0 or 1
  doc["airCtrl"]    = airCtrlState;    // 0 or 1
  doc["mode"]       = modeString;      // "AUTO" or "MANUAL"

  String jsonBody;
  serializeJson(doc, jsonBody);

  int httpCode = http.POST(jsonBody);
  http.end();
}

// ─── CALL IT IN YOUR MAIN loop() ───────────────────────────────
void loop() {
  // ... your existing sensor reading code ...
  
  // After reading sensors, send to server
  sendDataToServer();   // add this line
  
  delay(2000);           // 2 second loop
}
⚠️ Find your laptop IP: Open Command Prompt → type ipconfig → look for IPv4 Address under your Wi-Fi adapter (e.g. 192.168.1.5). Use that in the ESP32 code above.

Python — Live Data Server (ai_api.py upgrade)
REPLACE OR EXTEND YOUR EXISTING ai_api.py WITH THIS
PythonFlaskPort 5001
This file does 3 things: (1) Receives JSON from ESP32 at /data, (2) Serves live data to the website at /live, (3) Runs AI predictions at /predict.
Python# live_server.py — Run this on your laptop
# Start with: python live_server.py

from flask import Flask, request, jsonify
from flask_cors import CORS
import joblib
import numpy as np
from datetime import datetime

app = Flask(__name__)
CORS(app)  # Allow website/app to fetch data

# ─── LATEST DATA STORAGE (in memory) ───────────────────────────
latest_data = {
    "aqi": 0, "co2": 400, "temp": 25, "humidity": 50,
    "nh3": 0, "benzene": 0, "co": 0,
    "voltage": 0, "current": 0, "power": 0,
    "energy": 0, "cost": 0,
    "occupancy": 0, "airCtrl": 0, "mode": "AUTO",
    "timestamp": ""
}

# ─── LOAD YOUR TRAINED AI MODELS ───────────────────────────────
try:
    model_aqi_10  = joblib.load("model_aqi_10sec.pkl")
    model_aqi_30  = joblib.load("model_aqi_30sec.pkl")
    model_aqi_60  = joblib.load("model_aqi_60sec.pkl")
    model_co2_10  = joblib.load("model_co2_10sec.pkl")
    model_co2_30  = joblib.load("model_co2_30sec.pkl")
    model_co2_60  = joblib.load("model_co2_60sec.pkl")
    model_action  = joblib.load("model_action.pkl")
    model_anomaly = joblib.load("model_anomaly.pkl")
    models_loaded = True
    print("✅ All AI models loaded")
except Exception as e:
    models_loaded = False
    print(f"⚠️  Models not found: {e}")

# ─── ROUTE 1: ESP32 POSTS SENSOR DATA HERE ─────────────────────
@app.route('/data', methods=['POST'])
def receive_data():
    global latest_data
    data = request.get_json()
    if data:
        latest_data = data
        latest_data["timestamp"] = datetime.now().isoformat()
        print(f"📡 ESP32 data: AQI={data.get('aqi','?')} CO2={data.get('co2','?')}")
    return jsonify({"status": "ok"})

# ─── ROUTE 2: WEBSITE FETCHES LIVE DATA FROM HERE ──────────────
@app.route('/live', methods=['GET'])
def get_live():
    return jsonify(latest_data)

# ─── ROUTE 3: AI PREDICTIONS ───────────────────────────────────
@app.route('/predict', methods=['GET'])
def predict():
    d = latest_data
    features = np.array([[
        d["aqi"], d["co2"], d["temp"], d["humidity"],
        d["nh3"], d["benzene"], d["co"]
    ]])

    if not models_loaded:
        return jsonify({
            "aqi_now": d["aqi"], "aqi_10sec": d["aqi"],
            "aqi_30sec": d["aqi"], "aqi_60sec": d["aqi"],
            "co2_now": d["co2"], "co2_10sec": d["co2"],
            "co2_30sec": d["co2"], "co2_60sec": d["co2"],
            "action": "KEEP_OFF", "anomaly": "NORMAL"
        })

    return jsonify({
        "aqi_now":    float(d["aqi"]),
        "aqi_10sec":  float(model_aqi_10.predict(features)[0]),
        "aqi_30sec":  float(model_aqi_30.predict(features)[0]),
        "aqi_60sec":  float(model_aqi_60.predict(features)[0]),
        "co2_now":    float(d["co2"]),
        "co2_10sec":  float(model_co2_10.predict(features)[0]),
        "co2_30sec":  float(model_co2_30.predict(features)[0]),
        "co2_60sec":  float(model_co2_60.predict(features)[0]),
        "action":     str(model_action.predict(features)[0]),
        "anomaly":    "ANOMALY" if model_anomaly.predict(features)[0] == 1 else "NORMAL"
    })

# ─── START SERVER ──────────────────────────────────────────────
if __name__ == '__main__':
    print("🚀 AirMind server starting...")
    print("   ESP32 POST  → http://<YOUR_IP>:5001/data")
    print("   Website GET → http://<YOUR_IP>:5001/live")
    print("   AI GET      → http://<YOUR_IP>:5001/predict")
    app.run(host="0.0.0.0", port=5001, debug=True)

Update the Dashboard with Your Laptop IP
ONE LINE CHANGE IN dashboard.html AND mobile_app.html
The dashboard already calls these URLs. Just change the IP from 127.0.0.1 to your laptop's IP if opening from another device (like your phone).
dashboard.html — JavaScript// Find this section near the bottom of dashboard.html:

async function fetchLive() {
  try {
    // ↓ CHANGE THIS IP if opening from mobile or another PC
    const r = await fetch('http://127.0.0.1:5001/live');
    //               ↑ use 127.0.0.1 if opened on SAME laptop
    //               ↑ use 192.168.1.5 if opened on phone/other PC
    return await r.json();
  } catch { return mockLive(); }   // falls back to demo data
}

async function fetchAI() {
  try {
    const r = await fetch('http://127.0.0.1:5001/predict');
    return await r.json();
  } catch { return mockAI(); }
}
